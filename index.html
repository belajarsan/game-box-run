const dino = document.getElementById("dino"); const cactus = document.getElementById("cactus"); const gameOverText = document.getElementById("game-over-text"); const scoreElement = document.getElementById("score"); let isGameOver = false; let scoreInterval; // Fungsi Lompat function jump() { if (!dino.classList.contains("jump") && !isGameOver) { dino.classList.add("jump"); setTimeout(function () { dino.classList.remove("jump"); }, 300); } } // Fungsi Restart Game function restartGame() { isGameOver = false; gameOverText.style.display = "none"; cactus.classList.add("animate-cactus"); scoreElement.innerText = "Score: 0"; startScore(); } // Deteksi Tombol Spasi di Keyboard document.addEventListener("keydown", function (event) { if (event.code === "Space") { event.preventDefault(); // Mencegah layar scroll ke bawah saat tekan spasi if (isGameOver) { restartGame(); } else { jump(); } } }); // Sistem Skor function startScore() { let score = 0; scoreInterval = setInterval(() => { if (!isGameOver) { score++; scoreElement.innerText = "Score: " + score; } }, 200); } // Deteksi Tabrakan function checkCollision() { let dinoBottom = parseInt(window.getComputedStyle(dino).getPropertyValue("bottom")); let cactusLeft = parseInt(window.getComputedStyle(cactus).getPropertyValue("left")); if (cactusLeft > 40 && cactusLeft < 90 && dinoBottom <= 0) { isGameOver = true; clearInterval(scoreInterval); cactus.classList.remove("animate-cactus"); gameOverText.style.display = "block"; } } // Mulai Cek Tabrakan Terus-menerus setInterval(checkCollision, 10); startScore(); // Tambahkan ini agar bisa dimainkan dengan sentuhan di HP document.addEventListener("touchstart", function() { jump(); }); // Atau jika ingin bisa diklik pakai mouse juga di komputer: document.addEventListener("click", function() { jump(); });
